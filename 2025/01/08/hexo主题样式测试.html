import numpy as np
import pandas as pd

class DataProcessor:
    def __init__(self, data):
        self.data = data
        
    def clean_data(self):
        """清理数据的方法"""
        # 去除空值
        self.data = self.data.dropna()
        
        # 标准化数值列
        numeric_columns = self.data.select_dtypes(include=[np.number]).columns
        self.data[numeric_columns] = (self.data[numeric_columns] - 
                                     self.data[numeric_columns].mean()) / self.data[numeric_columns].std()
        
        return self.data
    
    def analyze(self):
        """数据分析方法"""
        return {
            'shape': self.data.shape,
            'description': self.data.describe(),
            'info': self.data.info()
        }

# 使用示例
processor = DataProcessor(pd.read_csv('data.csv'))
cleaned_data = processor.clean_data()
analysis_result = processor.analyze()

-- 复杂的数据分析查询
WITH monthly_sales AS (
    SELECT 
        DATE_TRUNC('month', order_date) as month,
        category,
        SUM(amount) as total_sales,
        COUNT(*) as order_count,
        AVG(amount) as avg_order_value
    FROM orders o
    JOIN products p ON o.product_id = p.id
    WHERE order_date >= '2024-01-01'
    GROUP BY DATE_TRUNC('month', order_date), category
),
category_growth AS (
    SELECT 
        month,
        category,
        total_sales,
        LAG(total_sales) OVER (PARTITION BY category ORDER BY month) as prev_month_sales,
        ROUND(
            ((total_sales - LAG(total_sales) OVER (PARTITION BY category ORDER BY month)) 
             / LAG(total_sales) OVER (PARTITION BY category ORDER BY month)) * 100, 2
        ) as growth_rate
    FROM monthly_sales
)

SELECT 
    month,
    category,
    total_sales,
    prev_month_sales,
    growth_rate,
    CASE 
        WHEN growth_rate > 20 THEN '高增长'
        WHEN growth_rate > 0 THEN '正增长'
        WHEN growth_rate < -10 THEN '显著下降'
        ELSE '稳定'
    END as growth_status
FROM category_growth
WHERE prev_month_sales IS NOT NULL
ORDER BY month DESC, growth_rate DESC;
